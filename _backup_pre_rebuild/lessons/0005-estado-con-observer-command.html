Lección 5: State con Observer y Command

Notificar cambios de estado y hacer transiciones deshacibles

Cuando el estado cambia el mundo

Un pedido pasa de Pendiente a Pagado, y el sistema notifica a envíos, facturación y permite deshacer la transición.

Python 3 · Estándar · 25 min

¿Por qué esta lección ahora?

State rara vez viaja solo. En el mundo real, cambiar de estado debe avisar a otros módulos y, a veces, deshacerse. Esta lección muestra cómo combinar tres patrones de comportamiento sin sobre-ingeniería.

1. Observer: avisar cuando cambia el estado

El Context mantiene una lista de observadores y los notifica cada vez que transita a un nuevo estado.

from __future__ import annotations
from abc import ABC, abstractmethod
from typing import List


# ---- Observer: suscriptores al cambio de estado ----
class ObservadorPedido(ABC):
    @abstractmethod
    def actualizar(self, pedido: "Pedido", estado_anterior: str, estado_nuevo: str) -> None:
        ...


class NotificadorEmail(ObservadorPedido):
    def actualizar(self, pedido, estado_anterior, estado_nuevo) -> None:
        print(f"[Email] Pedido {pedido.id}: {estado_anterior} -> {estado_nuevo}")


class LogAuditor(ObservadorPedido):
    def actualizar(self, pedido, estado_anterior, estado_nuevo) -> None:
        print(f"[Auditoria] {pedido.id} cambio de {estado_anterior} a {estado_nuevo}")


# ---- State: estados del pedido ----
class EstadoPedido(ABC):
    @abstractmethod
    def pagar(self, pedido: "Pedido") -> None: ...
    @abstractmethod
    def enviar(self, pedido: "Pedido") -> None: ...


class Pendiente(EstadoPedido):
    def pagar(self, pedido: "Pedido") -> None:
        pedido.cambiar_estado(Pagado())
    def enviar(self, pedido: "Pedido") -> None:
        print("No se puede enviar: aún no se ha pagado.")


class Pagado(EstadoPedido):
    def pagar(self, pedido: "Pedido") -> None:
        print("Ya está pagado.")
    def enviar(self, pedido: "Pedido") -> None:
        pedido.cambiar_estado(Enviado())


class Enviado(EstadoPedido):
    def pagar(self, pedido: "Pedido") -> None:
        print("Ya se envió.")
    def enviar(self, pedido: "Pedido") -> None:
        print("Ya se envió previamente.")


# ---- Context: Pedido con Observer embebido ----
class Pedido:
    def __init__(self, id_pedido: str) -> None:
        self.id = id_pedido
        self.estado: EstadoPedido = Pendiente()
        self._observadores: List[ObservadorPedido] = []
        self._estado_anterior: str = ""

    def agregar_observador(self, observador: ObservadorPedido) -> None:
        self._observadores.append(observador)

    def cambiar_estado(self, nuevo_estado: EstadoPedido) -> None:
        nombre_anterior = type(self.estado).__name__
        nombre_nuevo = type(nuevo_estado).__name__
        self._estado_anterior = nombre_anterior
        self.estado = nuevo_estado
        for obs in self._observadores:
            obs.actualizar(self, nombre_anterior, nombre_nuevo)

    def pagar(self) -> None:
        self.estado.pagar(self)

    def enviar(self) -> None:
        self.estado.enviar(self)


# ---- Uso ----
if __name__ == "__main__":
    p = Pedido("ORD-001")
    p.agregar_observador(NotificadorEmail())
    p.agregar_observador(LogAuditor())

    print("=== Transición 1 ===")
    p.pagar()   # Pendiente → Pagado (notifica a ambos observadores)

    print("\n=== Transición 2 ===")
    p.enviar()  # Pagado → Enviado (notifica a ambos observadores)

2. Command: transiciones deshacibles

Encapsula cada cambio de estado como un objeto comando, lo que permite registrar el historial y deshacer la última transición.

from __future__ import annotations
from abc import ABC, abstractmethod
from typing import List


# ---- Command: interfaz de transición ----
class ComandoTransicion(ABC):
    @abstractmethod
    def ejecutar(self) -> None: ...
    @abstractmethod
    def deshacer(self) -> None: ...


class ComandoPagar(ComandoTransicion):
    def __init__(self, pedido: "Pedido", estado_anterior: str) -> None:
        self.pedido = pedido
        self.estado_anterior = estado_anterior
        self.estado_nuevo = type(pedido.estado).__name__

    def ejecutar(self) -> None:
        self.pedido.pagar()

    def deshacer(self) -> None:
        # Restaura el estado anterior de forma simplificada
        mapa = {"Pendiente": Pendiente(), "Pagado": Pagado(), "Enviado": Enviado()}
        self.pedido.estado = mapa[self.estado_anterior]
        print(f"[Deshacer] {self.pedido.id} volvió a {self.estado_anterior}")


# ---- Historial de comandos ----
class HistorialTransiciones:
    def __init__(self) -> None:
        self._comandos: List[ComandoTransicion] = []

    def agregar(self, comando: ComandoTransicion) -> None:
        self._comandos.append(comando)

    def deshacer_ultimo(self) -> None:
        if not self._comandos:
            print("No hay nada que deshacer")
            return
        comando = self._comandos.pop()
        comando.deshacer()

3. ¿Cuándo usar cada combinación?

EscenarioPatronesPor qué
Solo necesitas cambiar comportamiento por estado State Mínimo acoplamiento, código limpio.
Otros módulos deben enterarse del cambio de estado State + Observer El contexto notifica sin conocer quién escucha.
Necesitas deshacer la última transición de estado State + Command Cada transición es un objeto reversible.
Notificar y deshacer al mismo tiempo State + Observer + Command Control total del ciclo de vida con historial.
